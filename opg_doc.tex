\begin{abstract}
This document is a rigorous, math-first description of the current model forward pass and training loss as implemented
in \texttt{train\_gpt.py}. The core clarification is that the model contains \emph{two distinct iteration axes}:
(i) a \textbf{self-refinement loop} that feeds predicted token distributions back into the \emph{input embedding}, and
(ii) a \textbf{RevDEQ coupled-state loop} that iterates the \emph{hidden state} to approach a fixed point of a learned update
function. The specification below is intended to be self-contained and auditable against the code.
\end{abstract}

\section{Practical defaults (throughput on 2$\times$A100 DDP)}
\label{sec:throughput}
\paragraph{Goal.}
For local iteration on this dev box (2$\times$A100 with DDP), the code defaults are chosen to maximize
\emph{tokens/second} while keeping behavior identical to the training/eval pipeline.

\paragraph{Defaults (as of 2026-04-06).}
\begin{itemize}
  \item \texttt{train\_batch\_tokens = 65536} (conservative throughput default on 2$\times$A100; larger batches require stability retuning).
  \item \texttt{compile\_train = 0} by default: \texttt{torch.compile} was empirically much slower for the shared RevDEQ block on 2$\times$A100.
  \item K-jitter uses a shuffle-bag over \texttt{[deq\_k\_min, deq\_k\_max]} with \textbf{no ramp} by default (stable from step 1).
\end{itemize}

\paragraph{Optional compile path.}
\texttt{compile\_train=1} remains supported for experimentation, but is not the throughput default on 2$\times$A100.

\section{Notation (shapes and conventions)}
\paragraph{Discrete inputs and targets.}
Let $V$ be the vocabulary size, $B$ the batch size, and $T$ the sequence length.
Inputs and targets are integer token IDs:
\[
  x \in \{0,\dots,V-1\}^{B\times T},\qquad
  y \in \{0,\dots,V-1\}^{B\times T}.
\]
In training, $y$ is the next-token target corresponding to $x$ (data windowing/stride is handled outside the model).

\paragraph{Embeddings and states.}
Let $d$ be the model dimension and $E\in\mathbb{R}^{V\times d}$ be the token embedding matrix.
The model operates on:
\[
  x_0 \in \mathbb{R}^{B\times T\times d}\quad\text{(input embedding to the backbone)},\qquad
  z \in \mathbb{R}^{B\times T\times d}\quad\text{(hidden state)}.
\]

\paragraph{Distributions.}
The model produces two log-probability tensors:
\[
  \log p_{\text{CTP}} \in \mathbb{R}^{B\times T\times V},\qquad
  \log p_{\text{NTP}} \in \mathbb{R}^{B\times T\times V},
\]
which are consumed in two ways:
\begin{itemize}
  \item for training losses, they are used directly in a negative log-likelihood;
  \item for self-refinement, the code converts them into \emph{valid} probability distributions by
    exponentiating and renormalizing in fp32.
\end{itemize}

\paragraph{Two iteration axes.}
We distinguish:
\begin{itemize}
  \item \textbf{Self-refinement iteration} $r\in\{0,1,\dots,R\}$ where $R=\texttt{num\_refinements}$.
    This is the outer loop: \emph{solve $\rightarrow$ predict $\rightarrow$ soft-embed $\rightarrow$ re-solve}.
  \item \textbf{DEQ iteration} $k\in\{0,1,\dots,K-1\}$ where $K=\texttt{num\_layers}$.
    This is the inner loop: coupled-state iteration that repeatedly applies a learned update function.
\end{itemize}

\section{End-to-end forward algorithm (single source of truth)}
\label{sec:forward}

\paragraph{Overview.}
The model maps token IDs $x$ to final hidden states $z^{(R)}$ via a backbone solve that may be repeated for refinement.
It then maps $z^{(R)}$ to two distributions (CTP and NTP) via a dual \textbf{MoSHead}.
Training losses are computed \emph{only on the final output} after the last refinement solve.

\paragraph{Pseudocode (matches \texttt{GPT.\_encode} + \texttt{GPT.\_run\_backbone}).}
\begin{lstlisting}
Inputs: token IDs x [B,T], targets y [B,T]
Hyperparams: R=num_refinements, K=num_layers, beta=deq_beta

# 1) Base input embedding (SmearGate is identity / removed)
x0^(0) = RMSNorm( Emb(x) + BigramHashEmb(x) )   # BigramHashEmb may be disabled
prev_soft = x0^(0)
z_init = x0^(0)

for r = 0..R:
  if r > 0:
    # No-grad: predict -> prob dist -> soft embed
    p_mix = MixProb( p_CTP(z^(r-1)), Shift(p_NTP(z^(r-1))) )
    soft = SoftEmbedTopK(p_mix, E)             # RMSNorm applied inside
    x0^(r) = 0.5 * soft + 0.5 * prev_soft      # EMA coupling
    prev_soft = stopgrad(x0^(r))
    z_init = x0^(r)                            # warm start from refined input
  else:
    x0^(0) already defined

  z^(r) = RevDEQ_Solve( f_theta, x0^(r), z_init, beta, K )

# 2) Final head
h = RMSNorm(z^(R))
log p_CTP, log p_NTP = MoSHead(h)
loss = NLL(log p_NTP, y) + ... (aux losses)
\end{lstlisting}

\section{Outer loop: self-refinement over input embeddings ($r$)}
\label{sec:refinement}

\subsection{Initial input embedding ($r=0$)}
The initial embedding is:
\[
  x_0^{(0)} \;=\; \mathrm{RMSNorm}\!\left(E[x] + \mathrm{BigramHashEmb}(x)\right),
\]
where \texttt{BigramHashEmb} may be disabled (then it contributes $0$).

\subsection{From predictions to a valid refinement distribution ($r>0$)}
At the end of solve $r-1$, define $h^{(r-1)}=\mathrm{RMSNorm}(z^{(r-1)})$ and compute:
\[
  \log p_{\text{CTP}}^{(r-1)},\ \log p_{\text{NTP}}^{(r-1)} \;=\; \mathrm{MoSHead}(h^{(r-1)}).
\]
For self-refinement, the code converts each log-prob into a numerically stable probability distribution:
\[
  \mathrm{Prob}(\log p)[v] \;=\;
  \frac{\exp(\log p[v])}{\sum_{u=1}^{V}\exp(\log p[u])}\quad\text{(computed in fp32, then renormalized)}.
\]
Let $p_{\text{CTP}}^{(r-1)}=\mathrm{Prob}(\log p_{\text{CTP}}^{(r-1)})$.
Define a shifted next-token distribution so that position $t$ uses the previous token's next-token predictor:
\[
  p_{\text{NTP,shift}}^{(r-1)}[t] \;\equiv\;
  \begin{cases}
    p_{\text{NTP}}^{(r-1)}[0] & t=0,\\
    p_{\text{NTP}}^{(r-1)}[t-1] & t\ge 1.
  \end{cases}
\]
Then the mixed refinement distribution is:
\[
  p_{\text{mix}}^{(r-1)}[t] \;=\;
  \begin{cases}
    p_{\text{CTP}}^{(r-1)}[0] & t=0,\\
    \tfrac{1}{2}p_{\text{CTP}}^{(r-1)}[t] + \tfrac{1}{2}p_{\text{NTP,shift}}^{(r-1)}[t] & t\ge 1.
  \end{cases}
\]
Because both inputs are valid distributions, $p_{\text{mix}}$ is also a valid distribution in exact arithmetic.
The implementation still applies NaN/Inf guarding.

\subsection{Soft embedding from a top-$\kappa$ approximation}
\label{sec:softembed}
The code approximates the expected embedding under $p_{\text{mix}}$ using a top-$\kappa$ truncation
(default $\kappa=128$; notation uses $\kappa$ to avoid confusion with the DEQ iteration count $K$).
Let $\mathcal{S}_\kappa(t)$ be the indices of the top-$\kappa$ probabilities at position $t$.
Define a renormalized distribution over $\mathcal{S}_\kappa(t)$:
\[
  \tilde p_{\text{mix}}[t,v] \;=\;
  \frac{p_{\text{mix}}[t,v]}{\sum_{u\in\mathcal{S}_\kappa(t)} p_{\text{mix}}[t,u]}
  \qquad (v\in\mathcal{S}_\kappa(t)).
\]
The soft embedding is the expected embedding under $\tilde p_{\text{mix}}$:
\[
  x_{\text{soft}}^{(r)}[t] \;=\; \mathrm{RMSNorm}\!\left(\sum_{v\in\mathcal{S}_\kappa(t)} \tilde p_{\text{mix}}^{(r-1)}[t,v]\ E[v]\right).
\]
Implementation note: the expected embedding is computed in chunks to avoid materializing a large $[B,T,\kappa,d]$ tensor.

\subsection{EMA coupling (refinement state update)}
Refinement updates the \emph{input embedding} for the next solve via an EMA:
\[
  x_0^{(r)} \;=\; \tfrac{1}{2}x_{\text{soft}}^{(r)} + \tfrac{1}{2}x_0^{(r-1)}.
\]
The previous refined input is detached in code, so gradients do not flow through earlier refinement steps.

\paragraph{One-hot as a special case.}
The $r=0$ embedding path corresponds to a delta distribution at each position
($p(v)=\mathbb{1}[v=x_t]$), whose expected embedding is exactly $E[x_t]$.

\section{Inner loop: RevDEQ coupled-state solver over hidden states ($k$)}
\label{sec:deq}

\subsection{Coupled-state fixed-\texorpdfstring{$K$}{K} iteration (explicit unroll)}
Within a single refinement solve $r$, the backbone runs $K$ coupled-state iterations with relaxation parameter $\beta\in(0,1)$.
It maintains two states $y_k^{(r)}$ and $z_k^{(r)}$ (both in $\mathbb{R}^{B\times T\times d}$) and applies $f_\theta$ twice per step:
\begin{align}
  y_{k+1}^{(r)} &= (1-\beta)\,y_k^{(r)} + \beta\, f_\theta\!\left(z_k^{(r)}, x_0^{(r)}\right),\\
  z_{k+1}^{(r)} &= (1-\beta)\,z_k^{(r)} + \beta\, f_\theta\!\left(y_{k+1}^{(r)}, x_0^{(r)}\right),
\end{align}
for $k=0,\dots,K-1$.

\paragraph{Warm start and terminal output.}
The code uses $y_0^{(r)}=z_0^{(r)}=z_{\text{init}}^{(r)}$ and sets $z_{\text{init}}^{(r)}=x_0^{(r)}$.
The solve returns the terminal state $z^{(r)}\equiv z_K^{(r)}$ and also tracks $z_{K-1}^{(r)}$ for convergence diagnostics.

\subsection{Backward modes (same forward)}
The forward dynamics above are the same for both backward modes:
\begin{itemize}
  \item \texttt{deq\_backward=autograd}: explicit unroll retains activations and supports differentiable output-space regularizers.
  \item \texttt{deq\_backward=revdeq}: uses a custom memory-efficient backward (\texttt{RevDEQFunction}) that reconstructs states in reverse.
    FP64 add/sub is used inside reconstruction for numerical stability; reconstruction error is logged only for this mode.
\end{itemize}

\section{Update function \texorpdfstring{$f_\theta(z,x_0)$}{f(z,x0)} (Block)}
\label{sec:f_theta}

\subsection{Input injection via a pooled scalar gate}
The block injects $x_0$ into the current state using a \emph{pooled scalar} convex gate.
Let $z_{\text{in}}$ denote the pre-injection state (identity anchor for the DEQ update).
Define a pooled summary vector from normalized state:
\[
  u \;=\; \mathbb{E}_{b,t}[\mathrm{RMSNorm}(z_{\text{in}})[b,t,:]] \in \mathbb{R}^d.
\]
The injection gate is a learned scalar:
\[
  g_{\text{inj}} \;=\; \sigma(w_{\text{inj}}^\top u + b_{\text{inj}})\in(0,1),
\]
and injection is a convex update written additively:
\[
  x \;=\; z_{\text{in}} + g_{\text{inj}}(x_0 - z_{\text{in}}).
\]
This keeps the injection cost small and stable under DEQ iteration while anchoring the solve to the token embedding.

\subsection{Component-level dense routing (Soft Dense Routing)}
The block contains two dense MoE components: attention and MLP. Each has its own router.
For a router input $h\in\mathbb{R}^d$, routing weights are:
\[
  w(h) \;=\; \mathrm{softmax}(\ell(h))\in\Delta^{E-1},
\]
so $w\in\mathbb{R}_{\ge 0}^{E}$ and $\sum_{e=1}^{E} w_e = 1$ (pure softmax; no post-softmax sigmoid gating).
All experts process all tokens; the routed update is the convex mixture of expert outputs.

\subsection{Global residual gate for contraction}
The block computes a per-token global gate $g_g(x)\in(0,1)$ from the normalized attention input:
\[
  g_g(x)\;=\;\sigma(\mathrm{gg}(\mathrm{RMSNorm}(x)))\in(0,1).
\]
Let $z_2$ be the post-attention+MLP residual result. The block update is:
\[
  f_\theta(z_{\text{in}},x_0)\;=\;(1-g_g)\,z_{\text{in}} + g_g\,z_2.
\]
This provides an explicit, logged knob for fixed-point behavior; convergence is monitored but not a hard constraint.

\section{Heads and losses (training objective)}
\label{sec:loss}

\subsection{CTP and NTP log-probabilities}
After the final refinement solve ($r=R$), compute:
\[
  h=\mathrm{RMSNorm}(z^{(R)}),\qquad
  \log p_{\text{CTP}},\ \log p_{\text{NTP}} = \mathrm{MoSHead}(h).
\]
CTP is trained to predict the current token $x$, and NTP predicts the next token $y$.

\subsection{Main negative log-likelihood losses}
The script uses \texttt{F.nll\_loss} on the log-probabilities:
\[
  L_{\text{NTP}} = \mathrm{NLL}(\log p_{\text{NTP}}, y),\qquad
  L_{\text{CTP}} = \mathrm{NLL}(\log p_{\text{CTP}}, x).
\]
The CTP weight scales with refinement steps:
\[
  \lambda_{\text{CTP}} = 0.1\,R.
\]

\subsection{Router health loss (hard-constraint shaped)}
Each router defines a per-token share distribution $w\in\Delta^{E-1}$. Let $\bar w=\mathbb{E}_{b,t}[w_{b,t}]$.
The training-time router loss is a constraint-shaped combination:
\[
  L_{\text{bal}}=\left\|\bar w-\tfrac{1}{E}\mathbf{1}\right\|_2^2
  \;+\; \lambda_{\min}\,\mathbb{E}_{e}\big[\max(0, \tfrac{0.6}{E}-\bar w_e)^2\big]
  \;+\; \lambda_{\mathrm{cv}}\,\max(0, \mathrm{CV}(\bar w)-0.20)^2,
\]
where $\mathrm{CV}(\bar w)=\mathrm{std}(\bar w)/\mathrm{mean}(\bar w)$. These soft barriers target the final-only hard constraints
(min-share and CV) without over-regularizing already-healthy routing.
Additionally, a loss-free bias controller updates router logits outside autograd to counteract collapse.

\subsection{Output-space orthogonality penalties (autograd-unroll only)}
When \texttt{deq\_backward=autograd}, the model may add differentiable output-space orthogonality penalties for expert outputs.
The tracked statistic is a mean off-diagonal absolute cosine similarity across experts:
\[
  \mathrm{Ortho}(o) \;=\; \mathbb{E}_{e\ne e'}\left[\,\left|\cos(o^{(e)}, o^{(e')})\right|\,\right].
\]
The loss includes weighted orthogonality terms for attention/MLP experts and for the MoS head (CTP + NTP).

\subsection{Total loss returned by \texttt{GPT.forward}}
The model returns:
\[
  L \;=\; L_{\text{NTP}} + \lambda_{\text{CTP}}L_{\text{CTP}}
  + \lambda_{\text{bal}}L_{\text{bal}} + \lambda_{\text{spar}}L_{\text{spar}}
  + \lambda_{\text{mos}}L_{\text{mos-ortho}}
  + \lambda_{\text{attn}}L_{\text{attn-ortho}} + \lambda_{\text{mlp}}L_{\text{mlp-ortho}},
\]
where $L_{\text{spar}}$ is currently a zero placeholder in code (hook for sparsity losses, not actively used).

% Intermediate DEQ supervision is not used in the current training pipeline.
\[
  L_{\text{sup}} \;=\; \sum_{k\in\mathcal{K}_{\text{sup}}} \omega_k\ \mathrm{NLL}(\log p_{\text{NTP}}(z_k), y),
\]
and the returned loss includes $+\lambda_{\text{sup}}L_{\text{sup}}$.
